Swipe UI is done. It’s smooth and full-screen.
Also added tracking — it logs how long a user stays on each card and detects skip/view/engagement.

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: ReelScreen(),
    );
  }
}

class ReelScreen extends StatefulWidget {
  @override
  _ReelScreenState createState() => _ReelScreenState();
}

class _ReelScreenState extends State<ReelScreen> {
  final PageController _controller = PageController();
  final List<Color> colors = [
    Colors.red,
    Colors.blue,
    Colors.green,
    Colors.orange,
    Colors.purple,
  ];

  int currentPage = 0;
  late DateTime startTime;

  @override
  void initState() {
  super.initState();
  startTime = DateTime.now();
  }

  @override
  void dispose() {
  _controller.dispose();
  super.dispose();
  }

  final List<String> texts = [
    "Card 1 - Learn Arrays",
    "Card 2 - Learn Stacks",
    "Card 3 - Learn Queues",
    "Card 4 - Learn Trees",
    "Card 5 - Learn Graphs",
  ];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: PageView.builder(
        onPageChanged: (index) {
          final endTime = DateTime.now();
          final duration = endTime.difference(startTime);

          final seconds = duration.inSeconds;

          if (seconds <= 1) {
            print("User SKIPPED card $currentPage");
          } else if (seconds <= 5) {
            print("User VIEWED card $currentPage");
          } else {
            print("User ENGAGED deeply with card $currentPage");
          }

          currentPage = index % colors.length;
          startTime = DateTime.now();
        },
        controller: _controller,
        scrollDirection: Axis.vertical,
        physics: PageScrollPhysics(),
        itemCount: colors.length,  
        itemBuilder: (context, index) {
          final realIndex = index % colors.length;

          return Container(
            color: colors[realIndex],
            child: Padding(
              padding: const EdgeInsets.all(20.0),
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    "Topic ${realIndex + 1}",
                    style: TextStyle(
                    fontSize: 18,
                    color: Colors.white70,
                    ),
                  ),
                  SizedBox(height: 20),
                  Text(
                    texts[realIndex],
                    style: TextStyle(
                      fontSize: 30,
                    color: Colors.white,
                    fontWeight: FontWeight.bold,
                    ),
                  ),
                  SizedBox(height: 20),
                  Text(
                    "Swipe up to continue learning",
                    style: TextStyle(
                    fontSize: 16,
                    color: Colors.white60,
                    ),
                  ),
                ],
              ),
            ),
          );
        },
      ),
    );
  }
}
