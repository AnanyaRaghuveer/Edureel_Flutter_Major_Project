import 'package:flutter/material.dart';
import 'package:share_plus/share_plus.dart';
import 'package:screenshot/screenshot.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: HomeScreen(),
    );
  }
}

class HomeScreen extends StatefulWidget {
  @override
  _HomeScreenState createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  int selectedIndex = 0;

  final List<Widget> screens = [
    ReelScreen(),
    DashboardScreen(),
  ];

  void onItemTapped(int index) {
    setState(() {
      selectedIndex = index;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: screens[selectedIndex],
      bottomNavigationBar: BottomNavigationBar(
        currentIndex: selectedIndex,
        onTap: onItemTapped,
        items: const [
          BottomNavigationBarItem(
            icon: Icon(Icons.play_arrow),
            label: "Feed",
          ),
          BottomNavigationBarItem(
            icon: Icon(Icons.dashboard),
            label: "Dashboard",
          ),
        ],
      ),
    );
  }
}

class DashboardScreen extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(
        child: Text(
          "Dashboard coming soon",
          style: TextStyle(fontSize: 22),
        ),
      ),
    );
  }
}

class ReelScreen extends StatefulWidget {
  @override
  _ReelScreenState createState() => _ReelScreenState();
}

class _ReelScreenState extends State<ReelScreen> {


  final ScreenshotController screenshotController = ScreenshotController();

  void shareSlide() async {
    // For web, use captureFromWidget and share as bytes
    final image = await screenshotController.captureFromWidget(
      Material(
        child: Container(
          color: colors[currentPage],
          child: Padding(
            padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 40),
            child: Column(
              mainAxisAlignment: MainAxisAlignment.spaceBetween,
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      "Topic "+(currentPage+1).toString(),
                      style: const TextStyle(fontSize: 18, color: Colors.white70),
                    ),
                    const SizedBox(height: 20),
                    Text(
                      texts[currentPage],
                      style: const TextStyle(fontSize: 30, color: Colors.white, fontWeight: FontWeight.bold),
                    ),
                  ],
                ),
                Row(
                  mainAxisAlignment: MainAxisAlignment.spaceBetween,
                  children: [
                    const Text(
                      "Swipe up to continue",
                      style: TextStyle(fontSize: 16, color: Colors.white60),
                    ),
                    Icon(Icons.share, color: Colors.white),
                  ],
                ),
              ],
            ),
          ),
        ),
      ),
      delay: const Duration(milliseconds: 10),
    );
    await Share.shareXFiles([
      XFile.fromData(image, name: 'slide.png', mimeType: 'image/png')
    ], text: "Check this out 👇");
  }

  void printAnalytics() {
    // Convert to 1-based index for output
    Map<int, Map<String, int>> finalOutput = {};
    slideAnalytics.forEach((key, value) {
      finalOutput[key + 1] = value;
    });
    print("BACKEND FORMAT: ${getBackendData()}");
  }

  List<Map<String, dynamic>> getBackendData() {
    List<Map<String, dynamic>> data = [];

    slideAnalytics.forEach((key, value) {
      data.add({
        "slide_id": "slide_${key + 1}",
        "total_time": value["total_time"] ?? 0,
        "views": value["views"] ?? 0,
        "engaged": value["engaged"] ?? 0,
        "rewatch": value["rewatch"] ?? 0,
      });
    });

    return data;
  }

  final PageController _controller = PageController();

  final List<Color> colors = [
    Colors.red,
    Colors.blue,
    Colors.green,
    Colors.orange,
    Colors.purple,
  ];

  int currentPage = 0;
  late DateTime startTime;

  Map<int, Map<String, int>> slideAnalytics = {};

  final List<String> texts = [
    "Card 1 - Learn Arrays",
    "Card 2 - Learn Stacks",
    "Card 3 - Learn Queues",
    "Card 4 - Learn Trees",
    "Card 5 - Learn Graphs",
  ];

  @override
  void initState() {
    super.initState();
    startTime = DateTime.now();

    // ✅ FIX: initialize properly (no null issues)
    slideAnalytics[0] = {
      "total_time": 0,
      "views": 1,
      "engaged": 0,
      "rewatch": 0,
    };
  }

  @override
  void dispose() {
    final endTime = DateTime.now();
    final seconds = endTime.difference(startTime).inSeconds;

    // Save last slide time safely
    if (slideAnalytics.containsKey(currentPage)) {
      slideAnalytics[currentPage]!["total_time"] =
          (slideAnalytics[currentPage]!["total_time"] ?? 0) + seconds;
    }

    // Convert to 1-based indexing (for UI/heatmap)
    Map<int, Map<String, int>> finalOutput = {};
    slideAnalytics.forEach((key, value) {
      finalOutput[key + 1] = value;
    });

    print("BACKEND FORMAT: ${getBackendData()}");

    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.black,
      body: SafeArea(
        child: PageView.builder(
          controller: _controller,
          scrollDirection: Axis.vertical,
          physics: const PageScrollPhysics(),
          itemCount: colors.length,
          onPageChanged: (index) {
            final endTime = DateTime.now();
            final seconds = endTime.difference(startTime).inSeconds;

            // ✅ Update previous slide safely
            if (slideAnalytics.containsKey(currentPage)) {
              slideAnalytics[currentPage]!["total_time"] =
                  (slideAnalytics[currentPage]!["total_time"] ?? 0) + seconds;

              if (seconds > 1) {
                slideAnalytics[currentPage]!["engaged"] =
                    (slideAnalytics[currentPage]!["engaged"] ?? 0) + 1;
              }
            }

            // Move to next slide
            currentPage = index % colors.length;
            startTime = DateTime.now();

            // ✅ Initialize or update views
            if (slideAnalytics.containsKey(currentPage)) {
                int currentViews = (slideAnalytics[currentPage]!["views"] ?? 0) + 1;
                slideAnalytics[currentPage]!["views"] = currentViews;

                // ✅ REWATCH LOGIC
                if (currentViews > 1) {
                  slideAnalytics[currentPage]!["rewatch"] =
                      (slideAnalytics[currentPage]!["rewatch"] ?? 0) + 1;
                }
              } else {
              slideAnalytics[currentPage] = {
                "total_time": 0,
                "views": 1,
                "engaged": 0,
                "rewatch": 0,
              };
            }
            // Print analytics after every slide change
            printAnalytics();
          },
          itemBuilder: (context, index) {
            final realIndex = index % colors.length;

            return Container(
              color: colors[realIndex],
              child: Padding(
                padding: const EdgeInsets.symmetric(
                  horizontal: 20,
                  vertical: 40,
                ),
                child: Column(
                  mainAxisAlignment: MainAxisAlignment.spaceBetween,
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Text(
                          "Topic ${realIndex + 1}",
                          style: const TextStyle(
                            fontSize: 18,
                            color: Colors.white70,
                          ),
                        ),
                        const SizedBox(height: 20),
                        Text(
                          texts[realIndex],
                          style: const TextStyle(
                            fontSize: 30,
                            color: Colors.white,
                            fontWeight: FontWeight.bold,
                          ),
                        ),
                      ],
                    ),
                    Row(
                      mainAxisAlignment: MainAxisAlignment.spaceBetween,
                      children: [
                        const Text(
                          "Swipe up to continue",
                          style: TextStyle(
                            fontSize: 16,
                            color: Colors.white60,
                          ),
                        ),
                        IconButton(
                          icon: const Icon(Icons.share, color: Colors.white),
                          onPressed: () {
                            shareSlide();
                          },
                        ),
                      ],
                    ),
                  ],
                ),
              ),
                          
            );
          },
        ),
      ),
    );
  }
}
